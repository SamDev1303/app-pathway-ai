### Task 5.1: Author CI grep gate workflow

- [ ] **Step 1:** Write `.github/workflows/mara-grep-gate.yml`:

```yaml
name: MARA + Privacy Gate

on:
  pull_request:
    branches: [main]
  push:
    branches: [main]

jobs:
  mara-forbidden-strings:
    name: MARA-forbidden strings check
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Grep for forbidden migration-advice language
        run: |
          set +e
          FORBIDDEN='visa|migration|\bPR\b|MLTSSL|subclass|points test|post-study work|permanent residency'
          HITS=$(grep -rEn "$FORBIDDEN" web/src \
            --include='*.ts' --include='*.tsx' \
            | grep -vE "mara-disclaimer\.ts|chat-system-prompt\.ts" \
            || true)
          if [ -n "$HITS" ]; then
            echo "::error::MARA-forbidden strings found in web/src (excluding allowlisted files):"
            echo "$HITS"
            exit 1
          fi
          echo "OK — no forbidden strings in web/src"

  mara-number-placeholder:
    name: MARA number placeholder enforcement
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Fail if PENDING_FROM_UNIMATE on main after 2026-04-27
        run: |
          set +e
          HITS=$(grep -rn "PENDING_FROM_UNIMATE" web/src || true)
          GRACE_DEADLINE=$(date -d '2026-04-28 00:00:00' +%s 2>/dev/null || date -j -f "%Y-%m-%d %H:%M:%S" "2026-04-28 00:00:00" +%s)
          NOW=$(date +%s)
          if [ -n "$HITS" ] && [ "$NOW" -ge "$GRACE_DEADLINE" ]; then
            echo "::error::MARA number placeholder PENDING_FROM_UNIMATE found in web/src after 2026-04-27 grace deadline:"
            echo "$HITS"
            exit 1
          fi
          if [ -n "$HITS" ]; then
            echo "::warning::MARA number placeholder still present — grace window until 2026-04-27"
          fi
          echo "OK — placeholder gate honored"
```
